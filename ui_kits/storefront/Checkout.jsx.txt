function StateScreen({icon,tone,eyebrow,title,children,actions}){
  const {EmptyState}=window.VendraDesignSystem_f4f210;
  const n=icon&&icon.props&&icon.props.name;
  return <EmptyState icon={n&&icon.type===window.VendraDesignSystem_f4f210.Icon?n:icon} tone={{ok:'accent',bad:'error'}[tone]||'neutral'} eyebrow={eyebrow} title={title} body={children} actions={actions} headingLevel={1} style={{padding:0}}/>;
}
function Checkout({lang,go,snap,form,zone,day,slot,payFail,onBack,onPlaced}){
  const {Button,IconButton,Input,Card,Icon,Radio,Badge,PaymentCard,LineItem}=window.VendraDesignSystem_f4f210;const T=AG_DATA.t[lang];const {mobile,tablet}=useBP();
  const P=AG_DATA.payment;const [st,setSt]=React.useState('pay');useHeadingFocus(st);const [m,setM]=React.useState('online');const [last4,setLast4]=React.useState('');const [err,setErr]=React.useState('');const [copied,setCopied]=React.useState(false);const [busy,setBusy]=React.useState(false);
  const codOk=P.codZones.includes(zone.id);const total=snap.sub+snap.fee;const money=v=>AG_DATA.money(v,lang);
  const slotL=(AG_DATA.delivery.IR.slots.find(s=>s[0]===slot)||[])[lang==='fa'?2:1];
  const whenL=zone.sameDay?(day==='today'?T.today:T.tomorrow)+' · '+slotL:T.noSameDay;

  // Button shows a spinner (loading) while the order is being placed; online then hands over to the full-screen "processing" state while the bank page runs.
  const pay=()=>{if(busy)return;
    if(m==='card'&&AG_DATA.toLatin(last4).replace(/\D/g,'').length!==4){setErr(T.last4Err);AG_NAV.focusFirstInvalid();return;}
    setBusy(true);setTimeout(()=>{setBusy(false);place();},700);};
  const place=()=>{
    if(window.AG_TRACK)AG_TRACK.event('add_payment_info',{payment_type:m,value:total});
    if(m==='online'){setSt('processing');window.scrollTo(0,0);setTimeout(()=>{if(payFail)setSt('fail');else{setSt('ok');AG_TRACK.event('purchase',{transaction_id:P.newOrder,value:total,shipping:snap.fee,payment_type:m,items:snap.items.map(l=>AG_TRACK.item(AG_DATA.products.find(x=>x.id===l.id),l.qty,l.unit))});onPlaced();}},1800);return;}
    if(m==='wa')window.open(AG_DATA.contact.wa+'?text='+waText,'_blank');
    setSt('ok');AG_TRACK.event('purchase',{transaction_id:P.newOrder,value:total,shipping:snap.fee,payment_type:m,items:snap.items.map(l=>AG_TRACK.item(AG_DATA.products.find(x=>x.id===l.id),l.qty,l.unit))});onPlaced();window.scrollTo(0,0);
  };
  const wrap=ch=><main id="main" style={{maxWidth:'var(--container-max)',margin:'0 auto',padding:mobile?'40px 16px 24px':'88px var(--gutter) 40px'}}>{ch}</main>;
  if(st==='processing')return wrap(<StateScreen icon={<span style={{display:'inline-flex',animation:'ag-spin 1s linear infinite'}}><Icon name="loader-circle" size={36}/></span>} title={T.processing}><style>{'@keyframes ag-spin{to{transform:rotate(360deg)}}'}</style>{T.processingP}</StateScreen>);
  if(st==='fail')return wrap(<><StateScreen tone="bad" icon={<Icon name="circle-x" size={38}/>} eyebrow={T.stepPay} title={T.failH} actions={<><Button size="lg" iconStart="rotate-ccw" onClick={pay}>{T.tryAgain}</Button><Button size="lg" variant="secondary" onClick={()=>setSt('pay')}>{T.otherMethod}</Button></>}>
    <p style={{margin:'0 0 8px'}}>{T.failP}</p><p style={{margin:0,color:'var(--text-muted)',fontSize:14}}>{T.failKept} · {money(total)}</p></StateScreen>
    <div style={{textAlign:'center',marginTop:24}}><Button variant="ghost" iconStart="message-circle" onClick={()=>window.open(AG_DATA.contact.wa+'?text='+waText,'_blank')}>{T.needHelp}</Button></div></>);
  if(st==='ok'){
    const rows=[[T.orderNo,<span dir="ltr">{P.newOrder}</span>],[T.deliverTo,form.name+' · '+form.address],[T.when,whenL],[T.payWith,<Badge tone={m==='online'?'sage':m==='card'?'accent':'neutral'}>{T.status[m]}</Badge>],[T.total,money(total)]];
    return wrap(<><CheckoutSteps lang={lang} at={3}/>
    <div style={{display:'grid',gridTemplateColumns:tablet?'1fr':'1.1fr 1fr',gap:mobile?32:64,alignItems:'center'}}>
      <div style={{textAlign:tablet?'center':'start'}}>
        <div style={{width:88,height:104,margin:tablet?'0 auto 24px':'0 0 24px',borderRadius:'999px 999px 12px 12px',background:'var(--accent-soft)',color:'var(--text-on-soft)',display:'grid',placeItems:'center'}}><Icon name="flower-2" size={38}/></div>
        <div className="ag-eyebrow" style={{marginBottom:10}}>{T.orderPlaced}</div>
        <h1 className="ag-display" style={{fontSize:mobile?38:60,lineHeight:1.02,margin:'0 0 16px',textWrap:'balance'}}>{T.okH}</h1>
        <p style={{color:'var(--text-secondary)',lineHeight:1.6,margin:'0 0 28px',textWrap:'pretty'}}>{T.okP.replace('{p}',AG_DATA.faDigits(form.phone,lang))}</p>
        {m==='card'&&<p style={{display:'flex',gap:10,alignItems:'flex-start',background:'var(--surface-sunken)',borderRadius:'var(--radius-sm)',padding:'12px 14px',fontSize:14,color:'var(--text-secondary)',textAlign:'start',margin:'-8px 0 28px'}}><Icon name="clock" size={18} style={{marginTop:2,flex:'none'}}/>{T.cardNote}</p>}
        <div style={{display:'flex',gap:12,flexWrap:'wrap',justifyContent:tablet?'center':'flex-start',flexDirection:mobile?'column':'row'}}><Button size="lg" iconEnd="arrow-right" href={AG_NAV.href('track','VN-10491')} onClick={AG_NAV.link}>{T.trackCta}</Button><Button size="lg" variant="secondary" href={AG_NAV.href('shop')} onClick={AG_NAV.link}>{T.continueShop}</Button></div>
      </div>
      <Card variant="sunken" padding={mobile?20:32}>
        <div style={{paddingBottom:8,marginBottom:6,borderBottom:'1px solid var(--border-default)'}}>{snap.items.map((l,i)=>{const p=AG_DATA.products.find(x=>x.id===l.id);const sz=T.sizes.find(x=>x[0]===l.size);return <LineItem key={i} size="sm" image={p.image} name={p[lang]} meta={sz&&sz[1]} price={money(l.unit*l.qty)} quantity={l.qty} formatQuantity={n=>AG_DATA.num(n,lang)}/>;})}</div>
        {rows.map(([a,b])=><div key={a} style={{display:'grid',gridTemplateColumns:'110px 1fr',gap:12,padding:'12px 0',borderBottom:'1px solid var(--border-subtle)',fontSize:14}}><span style={{color:'var(--text-muted)'}}>{a}</span><span style={{textAlign:'end',minWidth:0}}>{b}</span></div>)}
      </Card>
    </div></>);
  }
  const methods=['online','card','cod','wa'];
  return wrap(<><CheckoutSteps lang={lang} at={1}/>
    <div style={{display:'grid',gridTemplateColumns:tablet?'1fr':'1.5fr 1fr',gap:mobile?28:56,alignItems:'start'}}>
      <div>
        <Button variant="ghost" size="sm" iconStart={lang==='fa'?'arrow-right':'arrow-left'} onClick={onBack} style={{marginBottom:12,marginInlineStart:-12}}>{T.backToBag}</Button>
        <h1 className="ag-display" style={{fontSize:mobile?34:52,margin:'0 0 24px'}}>{T.payH}</h1>
        <div role="radiogroup" style={{display:'flex',flexDirection:'column',gap:12}}>{methods.map(id=>{const on=m===id;const off=id==='cod'&&!codOk;return <div key={id} style={{border:'1px solid '+(on?'var(--text-body)':'var(--border-default)'),borderRadius:'var(--radius-md)',background:on?'var(--surface-card)':'transparent',opacity:off?.5:1,transition:'border-color .15s'}}>
          <div style={{padding:mobile?'14px 16px':'18px 20px'}}><Radio name="pay" checked={on} disabled={off} onChange={()=>{setM(id);setErr('');}} label={T.pm[id][0]} description={off?T.codOff:T.pm[id][1]}/></div>
          {on&&id==='online'&&<div style={{padding:mobile?'0 16px 16px':'0 20px 20px',fontSize:14,color:'var(--text-muted)',display:'flex',gap:10}}><Icon name="shield-check" size={18} style={{flex:'none',marginTop:1}}/>{T.onlineNote}</div>}
          {on&&id==='card'&&<div style={{padding:mobile?'0 16px 16px':'0 20px 20px',display:'flex',flexDirection:'column',gap:16}}>
            <PaymentCard cardNumber={P.card} holder={P.holder[lang]} bank={P.bank[lang]} amount={money(total)} labels={{card:T.cardTo,holder:T.cardHolder,bank:lang==='fa'?'بانک':'Bank',amount:T.total,copy:T.copy,copied:T.copied}}/>

            <div style={{display:'grid',gridTemplateColumns:mobile?'1fr':'1fr 1fr',gap:14}}><Input label={T.last4} value={last4} onChange={e=>{setLast4(e.target.value.slice(0,4));setErr('');}} inputMode="numeric" dir="ltr" placeholder="1234" error={err||undefined}/><Input label={T.trackNo} inputMode="numeric" dir="ltr"/></div>
            <div style={{fontSize:14,color:'var(--text-muted)',display:'flex',gap:10}}><Icon name="clock" size={18} style={{flex:'none',marginTop:1}}/>{T.cardNote}</div>
          </div>}
          {on&&id==='cod'&&<div style={{padding:mobile?'0 16px 16px':'0 20px 20px',fontSize:14,color:'var(--text-muted)',display:'flex',gap:10}}><Icon name="banknote" size={18} style={{flex:'none',marginTop:1}}/>{T.codNote}</div>}
          {on&&id==='wa'&&<div style={{padding:mobile?'0 16px 16px':'0 20px 20px',fontSize:14,color:'var(--text-muted)',display:'flex',gap:10}}><Icon name="message-circle" size={18} style={{flex:'none',marginTop:1}}/>{T.waNote}</div>}
        </div>;})}</div>
      </div>
      <Card variant="sunken" padding={mobile?20:32} style={{position:tablet?'static':'sticky',top:96}}>
        <div className="ag-display" style={{fontSize:mobile?26:30,marginBottom:16}}>{T.summary}</div>
        {[[T.deliverTo,form.name+' · '+AG_DATA.faDigits(form.phone,lang)],[T.address,form.address],[T.when,whenL]].map(([a,b])=><div key={a} style={{fontSize:14,marginBottom:12}}><div style={{color:'var(--text-muted)',fontSize:12,marginBottom:2}}>{a}</div><div style={{overflowWrap:'anywhere'}}>{b}</div></div>)}
        <div style={{borderTop:'1px solid var(--border-default)',paddingTop:14,marginTop:4}}>{[[T.subtotal,money(snap.sub)],[T.delivery,snap.fee===0?T.free:money(snap.fee)]].map(([a,b])=><div key={a} style={{display:'flex',justifyContent:'space-between',marginBottom:10,color:'var(--text-secondary)'}}><span>{a}</span><span>{b}</span></div>)}</div>
        <div style={{display:'flex',justifyContent:'space-between',padding:'14px 0 22px',borderTop:'1px solid var(--border-default)',fontSize:mobile?18:20,fontWeight:500,gap:12}}><span>{T.total}</span><span>{money(total)}</span></div>
        <Button block size="lg" loading={busy} loadingLabel={lang==='fa'?'در حال ثبت سفارش':'Placing your order'} iconStart={m==='wa'?'message-circle':m==='online'?'lock':undefined} onClick={pay}>{m==='online'?T.payBtn.replace('{a}',money(total)):m==='wa'?T.sendWa:T.placeBtn}</Button>
        {m==='online'&&<div style={{display:'flex',gap:6,justifyContent:'center',alignItems:'center',marginTop:12,fontSize:12,color:'var(--text-muted)'}}><Icon name="shield-check" size={14}/>{T.secure}</div>}
      </Card>
    </div></>);
}

function OrderLookup({lang,go,missing}){
  const {Button,Input,Icon}=window.VendraDesignSystem_f4f210;const T=AG_DATA.t[lang];const {mobile}=useBP();
  const [v,setV]=React.useState(missing||'');const [err,setErr]=React.useState(missing?T.lookErr.replace('{id}',missing):'');
  const find=e=>{e.preventDefault();const id=AG_DATA.toLatin(v).trim().toUpperCase().replace(/^(VN)?-?/,'VN-');if(AG_DATA.account.orders.some(o=>o.id===id))go('track',id);else{setErr(T.lookErr.replace('{id}',v.trim()||'—'));AG_NAV.focusFirstInvalid();}};
  return <main id="main" style={{maxWidth:480,margin:'0 auto',padding:mobile?'48px 16px 32px':'104px var(--gutter) 64px'}}>
    <StateScreen tone={err?'bad':'calm'} icon={<Icon name={err?'search-x':'package-search'} size={36}/>} eyebrow={T.account} title={T.lookH}>{T.lookP}</StateScreen>
    <form onSubmit={find} style={{display:'flex',flexDirection:'column',gap:14,marginTop:28}}>
      <Input label={T.orderNoL} value={v} onChange={e=>{setV(e.target.value);setErr('');}} dir="ltr" placeholder="VN-10491" error={err||undefined} hint={err?undefined:T.lookHint} autoFocus/>
      <Button type="submit" block size="lg">{T.lookBtn}</Button>
      {err&&<Button type="button" variant="ghost" block iconStart="message-circle" onClick={()=>window.open(AG_DATA.contact.wa,'_blank')}>{T.needHelp}</Button>}
    </form>
  </main>;
}

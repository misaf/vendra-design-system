function OtpInput({value,onChange,len=5}){
  const refs=React.useRef([]);
  const set=(i,v)=>{v=v.replace(/[^0-9۰-۹]/g,'').slice(-1);const a=value.split('');a[i]=v;const n=a.join('').slice(0,len);onChange(n);if(v&&i<len-1)refs.current[i+1].focus();};
  return <div dir="ltr" style={{display:'flex',gap:10,justifyContent:'center'}}>{Array.from({length:len}).map((_,i)=><input key={i} ref={el=>refs.current[i]=el} value={value[i]||''} onChange={e=>set(i,e.target.value)} onKeyDown={e=>{if(e.key==='Backspace'&&!value[i]&&i>0)refs.current[i-1].focus();}} inputMode="numeric" autoComplete={i===0?'one-time-code':'off'} aria-label={'Digit '+(i+1)} style={{width:52,height:60,textAlign:'center',fontSize:24,fontFamily:'var(--font-body)',color:'var(--text-body)',background:'var(--surface-card)',border:'1px solid '+(value[i]?'var(--border-strong)':'var(--border-default)'),borderRadius:'var(--radius-sm)',outline:'none'}}/>)}</div>;
}

function Account({lang,setLang,go,signedIn,setSignedIn}){

// ?view=track&id=<orderNo> — order page (noindex). Status → step via AG_DATA.orderStatus().
function OrderDetail({lang,id,go,signedIn,onReorder}){
  const {Button,Card,Icon,Alert,OrderTimeline,OrderSummary,DetailList}=window.VendraDesignSystem_f4f210;const T=AG_DATA.t[lang];const {mobile,tablet}=useBP();const DT=AG_DATES;
  const o=id&&AG_DATA.account.orders.find(x=>x.id===id);
  const wa=text=>AG_DATA.contact.wa+'?text='+encodeURIComponent(text);
  const back=signedIn&&<a href={AG_NAV.href('account')} onClick={AG_NAV.link} style={{display:'inline-flex',alignItems:'center',gap:6,fontSize:14,marginBottom:16}}><Icon name="arrow-left" size={16}/>{T.backAccount}</a>;
  if(!o)return <main id="main" style={{maxWidth:560,margin:'0 auto',padding:mobile?'48px 16px 32px':'104px var(--gutter) 64px'}}>
    {back}
    <StateScreen tone="calm" icon={<Icon name="package-search" size={36}/>} eyebrow={id?<>{T.orderNo} <span dir="ltr">{id}</span></>:T.account} title={T.nfH}
      actions={<><Button size="lg" href={AG_NAV.href('account')} onClick={AG_NAV.link}>{signedIn?T.yourOrders:T.signIn}</Button><Button size="lg" variant="secondary" iconStart="message-circle" href={wa(id?T.orderNo+' '+id:T.help)} target="_blank">{T.helpBtn}</Button></>}>{T.nfP}</StateScreen>
  </main>;
  const st=AG_DATA.orderStatus(o.status);const key=AG_DATA.orderKey(o);const cancelled=st==='cancelled';
  const loc=DT.locale(lang,lang==='fa'?'j':'g');const dg=v=>DT.digits(v,lang);
  const zone=AG_DATA.delivery.IR.zones.find(z=>z.id===o.zone)||{};const zoneName=(zone[lang]||'').split(' (')[0];
  const slot=o.slot&&AG_DATA.delivery.IR.slots.find(s=>s[0]===o.slot);
  const tx=v=>v==null?'':typeof v==='string'?v:(v[lang]||v.en);
  const money=v=>AG_DATA.money(v,lang);const sub=AG_DATA.orderSub(o);
  const gone=it=>{const p=AG_DATA.products.find(x=>x.id===it.id);return !p||p.badge==='soldout'||p.price==null;};const goneN=o.items.filter(gone).length;const allGone=goneN===o.items.length;
  const goneL=lang==='fa'?'دیگر موجود نیست':'No longer available';const reNote=lang==='fa'?(allGone?'هیچ‌کدام از گل‌های این سفارش دیگر موجود نیست.':'گل‌هایی که دیگر موجود نیستند به سبد اضافه نمی‌شوند. قیمت‌ها به نرخ امروز است.'):(allGone?'None of these flowers are available any more.':'Items no longer available won’t be added. Prices are today’s.');
  const whenV=!zone.sameDay||!o.deliveryDate?T.postNote:DT.fullDate(DT.fromIso(o.deliveryDate),loc,true)+(slot?' · '+slot[lang==='fa'?2:1]:'');
  const steps=AG_DATA.orderSteps.map((s,i)=>({label:T.status[s],time:o.times[i]?dg(o.times[i]):null}));
  return <main id="main" style={{maxWidth:1040,margin:'0 auto',padding:mobile?'24px 16px 0':'56px var(--gutter) 0'}}>
    {back}
    <div className="ag-eyebrow">{T.orderNo} <span dir="ltr">{o.id}</span> · {DT.fullDate(DT.fromIso(o.placed),loc)}</div>
    <h1 className="ag-display" style={{fontSize:mobile?40:60,margin:'8px 0 6px'}}>{AG_DATA.orderHeading(o,lang)}</h1>
    {!cancelled&&<p style={{color:'var(--text-secondary)',margin:'0 0 28px',fontSize:mobile?16:18}}>{T.statusP[key]}</p>}
    {cancelled&&<div style={{margin:'16px 0 28px'}}><Alert tone="danger">{T.statusP.cancelled}</Alert></div>}
    <div style={{display:'grid',gridTemplateColumns:tablet?'minmax(0,1fr)':'minmax(0,1.2fr) minmax(0,1fr)',gap:mobile?20:32,alignItems:'start'}}>
      <div style={{display:'flex',flexDirection:'column',gap:20,minWidth:0}}>
        {!cancelled&&<Card padding={mobile?20:28}><OrderTimeline label={T.progress} doneLabel={T.doneSr} steps={steps} current={st} status={st===4?'done':'active'}/></Card>}
        <Card variant="sunken" padding={mobile?20:24}><DetailList rows={[
          {icon:'map-pin',label:T.dAddr,value:tx(o.address.line)+(zoneName?' · '+zoneName:'')},
          {icon:'clock',label:T.dWhen,value:whenV},
          {icon:'user',label:T.dRecipient,value:<>{tx(o.address.recipient)} · <span dir="ltr">{o.address.phone}</span></>},
          {icon:'mail',label:T.dCard,value:o.msg?<span className="ag-display" style={{fontSize:20,fontStyle:lang==='fa'?'normal':'italic'}}>“{tx(o.msg)}”</span>:T.noCard},
          o.payRef&&{icon:'receipt',label:T.dPayRef,value:<span dir="ltr">{dg(o.payRef)}</span>}]}/></Card>
      </div>
      <div style={{display:'flex',flexDirection:'column',gap:20,position:tablet?'static':'sticky',top:96,minWidth:0}}>
        <Card padding={mobile?20:24}><OrderSummary title={T.yourOrder} titleAs="h2"
          lines={o.items.map(it=>{const p=AG_DATA.products.find(x=>x.id===it.id);return {name:p[lang],href:AG_NAV.href('product',p.id),onClick:AG_NAV.link,image:p.image,meta:'× '+dg(it.qty)+' · '+money(it.unit)+(gone(it)?' · '+goneL:''),total:money(it.unit*it.qty)};})}
          sums={[{label:T.subtotal,value:money(sub)},{label:T.delivery,value:o.fee?money(o.fee):T.free},{label:T.total,value:money(sub+o.fee),strong:true}]}/></Card>
        <div style={{display:'flex',flexDirection:'column',gap:10}}>
          <Button block variant="secondary" iconStart="message-circle" href={wa(T.orderNo+' '+o.id)} target="_blank">{T.askOrder}</Button>
          <Button block iconStart="rotate-ccw" disabled={allGone} onClick={()=>onReorder&&onReorder(o)}>{T.orderAgain}</Button>
          {goneN>0&&<p style={{margin:0,fontSize:14,color:'var(--text-secondary)'}}>{reNote}</p>}
        </div>
      </div>
    </div>
  </main>;
}
window.OrderDetail=OrderDetail;

function SearchOverlay({lang,open,onClose,go}){
  const {Icon,IconButton,Tag,EmptyState,ArchFrame}=window.VendraDesignSystem_f4f210;const T=AG_DATA.t[lang];const {mobile}=useBP();
  const [q,setQ]=React.useState('');const ref=React.useRef();
  React.useEffect(()=>{if(open){setQ('');setTimeout(()=>ref.current&&ref.current.focus(),50);}const k=e=>e.key==='Escape'&&onClose();window.addEventListener('keydown',k);return()=>window.removeEventListener('keydown',k);},[open]);
  if(!open)return null;
  const s=q.trim().toLowerCase();
  const res=s?AG_DATA.products.filter(p=>[p.en,p.fa,p.subEn,p.subFa,AG_DATA.catName(p.cat,'en'),AG_DATA.catName(p.cat,'fa')].some(x=>x.toLowerCase().includes(s))):[];
  const cats=s?AG_DATA.categories.filter(c=>c.en.toLowerCase().includes(s)||c.fa.includes(s)):[];
  const sugg=lang==='fa'?['رز','ارکیده','باکس','عروس','گیاه']:['Roses','Orchid','Box','Bridal','Palm'];
  return <div onClick={e=>{if(e.target===e.currentTarget)onClose();}} style={{position:'fixed',inset:0,zIndex:110,background:'var(--surface-overlay)',backdropFilter:'blur(6px)',WebkitBackdropFilter:'blur(6px)'}}>
    <div role="dialog" aria-label={lang==='fa'?'جستجو':'Search'} style={{background:'var(--surface-page)',maxHeight:mobile?'100%':'80vh',height:mobile?'100%':'auto',overflowY:'auto',boxShadow:'var(--shadow-lg)'}}>
      <div style={{maxWidth:880,margin:'0 auto',padding:mobile?'12px 16px 24px':'28px var(--gutter) 36px'}}>
        <div style={{display:'flex',alignItems:'center',gap:12,borderBottom:'1.5px solid var(--border-strong)',paddingBottom:10}}>
          <Icon name="search" size={22} color="var(--text-muted)"/>
          <input ref={ref} value={q} onChange={e=>setQ(e.target.value)} placeholder={lang==='fa'?'جستجوی گل، دسته یا مناسبت…':'Search flowers, categories…'} style={{flex:1,minWidth:0,border:0,outline:0,background:'transparent',font:'400 '+(mobile?'20px':'26px')+' var(--font-display)',color:'var(--text-body)',height:48}}/>
          <IconButton icon="x" label={lang==='fa'?'بستن':'Close'} onClick={onClose}/>
        </div>
        {!s&&<div style={{marginTop:20}}><div className="ag-eyebrow" style={{marginBottom:10}}>{lang==='fa'?'پیشنهادها':'Try'}</div><div style={{display:'flex',gap:8,flexWrap:'wrap'}}>{sugg.map(x=><Tag key={x} onClick={()=>setQ(x)}>{x}</Tag>)}</div></div>}
        {cats.length>0&&<div style={{display:'flex',gap:8,flexWrap:'wrap',marginTop:18}}>{cats.map(c=><Tag key={c.id} onClick={()=>{onClose();go('shopcat',c.id);}}>{c[lang]} →</Tag>)}</div>}
        {s&&<div style={{marginTop:18,display:'flex',flexDirection:'column'}}>
          {res.length?res.map(p=><a key={p.id} href={AG_NAV.href('product',p.id)} onClick={e=>{onClose();AG_NAV.link(e);}} style={{textDecoration:'none',display:'grid',gridTemplateColumns:'52px 1fr auto',gap:14,alignItems:'center',padding:'10px 4px',border:0,borderBottom:'1px solid var(--border-subtle)',background:'transparent',textAlign:'start',cursor:'pointer',color:'var(--text-body)',font:'inherit'}}>
            <ArchFrame size="thumb" tone="product" src={p.image} alt="" style={{width:52}}/>
            <span style={{minWidth:0}}><span className="ag-display" style={{display:'block',fontSize:20}}>{p[lang]}</span><span style={{fontSize:13,color:'var(--text-muted)'}}>{AG_DATA.catName(p.cat,lang)}</span></span>
            <span style={{fontSize:14,fontWeight:500,whiteSpace:'nowrap'}}>{AG_DATA.money(p.price,lang)}</span>
          </a>):<EmptyState icon="search-x" headingLevel={3} title={lang==='fa'?'نتیجه‌ای پیدا نشد':'No results'} body={lang==='fa'?'نام گل یا دسته دیگری را امتحان کنید.':'Try another flower or category.'} style={{padding:'24px 0'}}/>}
        </div>}
      </div>
    </div>
  </div>;
}
function Saved({lang,go,favs,toggleFav}){
  const {Button,Icon,EmptyState}=window.VendraDesignSystem_f4f210;const T=AG_DATA.t[lang];const {mobile}=useBP();
  const items=AG_DATA.products.filter(p=>favs.includes(p.id));
  return <main id="main" style={{maxWidth:'var(--container-max)',margin:'0 auto',padding:mobile?'24px 16px 0':'56px var(--gutter) 0'}}>
    <div className="ag-eyebrow">{T.account}</div>
    <h1 className="ag-display" style={{fontSize:mobile?40:64,margin:mobile?'8px 0 20px':'12px 0 36px'}}>{lang==='fa'?'ذخیره‌شده‌ها':'Saved'}{items.length>0&&<span style={{fontSize:'0.45em',color:'var(--text-muted)',marginInlineStart:12}}>{AG_DATA.num(items.length,lang)}</span>}</h1>
    {items.length?<ProductGrid lang={lang} items={items} go={go} favs={favs} toggleFav={toggleFav}/>
    :<EmptyState icon="heart" title={lang==='fa'?'هنوز چیزی ذخیره نکرده‌اید':'Nothing saved yet'} body={lang==='fa'?'روی قلب هر محصول بزنید تا اینجا ذخیره شود.':'Tap the heart on any piece to save it here.'} actions={<Button iconEnd={'arrow-right'} href={AG_NAV.href('shop')} onClick={AG_NAV.link}>{T.keep}</Button>} style={{padding:'40px 0'}}/>}
  </main>;
}
Object.assign(window,{SearchOverlay,Saved});

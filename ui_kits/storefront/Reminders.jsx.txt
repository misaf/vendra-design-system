// Stored as {cal:'j'|'g', m, d}. Old records with an ISO `date` are read as Gregorian; old {m,d} without cal were Jalali.
function normReminder(r){if(r.cal)return r;if(r.date){const dt=AG_DATES.fromIso(r.date);if(dt)return {...r,cal:'g',m:dt.getMonth()+1,d:dt.getDate()};}return {...r,cal:'j'};}
function Reminders({lang,list:extList,setList:setExt}){
  const {Button,Input,Select,Dialog,Tabs,Icon,Toast,ReminderRow,DatePicker}=window.VendraDesignSystem_f4f210;const T=AG_DATA.t[lang];const {mobile}=useBP();
  const D=AG_DATA;const DT=AG_DATES;const n=v=>D.num(v,lang);
  const [own,setOwn]=React.useState(()=>D.account.reminders.map(normReminder));
  const list=(extList||own).map(normReminder);const setList=setExt||setOwn;
  const [edit,setEdit]=React.useState(null);const [err,setErr]=React.useState('');const [toast,setToast]=React.useState('');
  const flash=m=>{setToast(m);setTimeout(()=>setToast(''),2400);};
  const occ=id=>D.occasions.find(o=>o.id===id)||D.occasions[D.occasions.length-1];
  const nm=r=>typeof r.name==='string'?r.name:(r.name[lang]||r.name.en);
  const rows=list.map(r=>{const nx=D.occasionNext(r);if(occ(r.occ).movable){const p=DT.parts(r.cal,nx.date);return {...r,m:p[1],d:p[2],...nx};}return {...r,...nx};}).sort((a,b)=>(b.on-a.on)||a.days-b.days);
  const when=d=>d===0?T.today:d===1?T.tomorrowL:T.inDays.replace('{n}',n(d));
  const otherCal=c=>c==='j'?'g':'j';
  const next=rows.find(r=>r.on&&r.days<=14);
  // "Send flowers" → AG_DATA.occasionDest; screens this kit doesn't have fall back to the shop.
  const dest=r=>{const [v,id]=D.occasionDest[r.occ]||['shop'];return (AG_NAV.routes||[]).includes(v)?AG_NAV.href(v,String(id)):AG_NAV.href('shop');};
  const blank=()=>({id:null,name:'',occ:'birthday',cal:lang==='fa'?'j':'g',m:lang==='fa'?7:10,d:1,before:3,channel:'sms',on:true});
  const open=r=>{setErr('');setEdit(r?{...r,name:nm(r)}:blank());};
  const save=()=>{if(!String(edit.name).trim()){setErr(T.req);AG_NAV.focusFirstInvalid(document.querySelector('[role="dialog"]'));return;}
    const r={...edit,name:{en:edit.name.trim(),fa:edit.name.trim()}};delete r.date;delete r.days;
    if(!r.id&&window.AG_TRACK)AG_TRACK.event('reminder_created',{occasion:r.occ,calendar:r.cal,days_before:r.before,channel:r.channel});
    setList(r.id?list.map(x=>x.id===r.id?r:x):[...list,{...r,id:'r'+Date.now()}]);setEdit(null);flash(T.remSaved);};
  const focusHead=()=>setTimeout(()=>{const h=document.getElementById('rem-h');h&&h.focus({preventScroll:true});},30);
  const del=r=>{setList(list.filter(x=>x.id!==r.id));flash(T.remDeleted);focusHead();};
  const fixed=edit&&occ(edit.occ).fixed;const movable=edit&&occ(edit.occ).movable;
  const L=s=>({reminderFor:T.remFor.replace('{name}',s),edit:T.editRemN.replace('{name}',s),delete:T.delRemN.replace('{name}',s),paused:T.paused,sendFlowers:T.sendFlowers});
  return <div>
    <div style={{display:'flex',justifyContent:'space-between',alignItems:mobile?'stretch':'flex-end',gap:16,marginBottom:20,flexDirection:mobile?'column':'row'}}>
      <div><h2 id="rem-h" tabIndex={-1} className="ag-display" style={{fontSize:mobile?28:36,margin:0}}>{T.remH}</h2><p style={{color:'var(--text-secondary)',margin:'6px 0 0',maxWidth:480,fontSize:15}}>{T.remP}</p></div>
      {list.length>0&&<Button iconStart="plus" onClick={()=>open(null)} style={{flex:'none'}}>{T.addRem}</Button>}
    </div>
    {next&&<div style={{background:'var(--surface-inverse)',color:'var(--text-on-inverse)',borderRadius:'var(--radius-md)',padding:mobile?18:24,display:'flex',gap:16,alignItems:mobile?'flex-start':'center',flexDirection:mobile?'column':'row',marginBottom:20}}>
      <div style={{width:48,height:56,borderRadius:'999px 999px 8px 8px',background:'var(--accent)',color:'var(--text-on-accent)',display:'grid',placeItems:'center',flex:'none'}}><Icon name={occ(next.occ).icon} size={22}/></div>
      <div style={{flex:1,minWidth:0}}><div className="ag-eyebrow" style={{color:'var(--text-accent)'}}>{T.comingUp}</div><div style={{marginTop:4,lineHeight:1.5}}>{T.comingP.replace('{name}',nm(next)).replace('{occ}',occ(next.occ)[lang]).replace('{when}',when(next.days))}</div></div>
      <Button variant="soft" iconEnd="arrow-right" href={dest(next)} onClick={AG_NAV.link} style={{flex:'none'}}>{T.sendFlowers}</Button>
    </div>}
    {!list.length?<div style={{border:'1.5px dashed var(--border-default)',borderRadius:'var(--radius-md)',padding:mobile?'40px 20px':'56px 24px'}}><StateScreen icon={<Icon name="calendar-heart" size={34}/>} title={T.remEmptyH} actions={<Button size="lg" iconStart="plus" onClick={()=>open(null)}>{T.addRem}</Button>}>{T.remEmptyP}</StateScreen></div>
    :<div style={{display:'flex',flexDirection:'column',gap:12}}>{rows.map(r=>{const o=occ(r.occ);const soon=r.on&&r.days<=7;const s=nm(r);
      return <ReminderRow key={r.id} name={s} day={n(r.date&&DT.parts(r.cal,r.date)[2]||r.d)} month={DT.monthNames(r.cal,lang)[r.m-1]} occasion={o[lang]} occasionIcon={o.icon}
        before={T.beforeOpt[r.before]} channel={r.channel} altDate={DT.dayMonth(r.date,DT.locale(lang,otherCal(r.cal)))} when={when(r.days)} soon={soon} on={r.on}
        onToggle={v=>setList(list.map(x=>x.id===r.id?{...x,on:v}:x))} onEdit={()=>open(r)} onDelete={()=>del(r)}
        sendHref={dest(r)} sendOnClick={AG_NAV.link} labels={L(s)}/>;})}</div>}
    <Dialog open={!!edit} title={edit&&edit.id?T.editRem:T.addRem} onClose={()=>setEdit(null)} footer={<><Button variant="ghost" onClick={()=>setEdit(null)}>{T.cancel}</Button><Button onClick={save}>{T.save}</Button></>}>
      {edit&&<div style={{display:'flex',flexDirection:'column',gap:14,paddingTop:8}}>
        <Input id="rem-name" label={T.remName} placeholder={T.remNamePh} value={edit.name} onChange={e=>{setErr('');setEdit({...edit,name:e.target.value});}} error={err||undefined}/>
        <Select id="rem-occ" label={T.occasion} value={edit.occ} onChange={e=>{const o=occ(e.target.value);let dt=o.fixed||{};if(o.movable){const c=edit.cal;const p=DT.parts(c,D.occasionNext({occ:o.id}).date);dt={cal:c,m:p[1],d:p[2]};}setEdit({...edit,occ:o.id,...dt});}} options={D.occasions.map(o=>({value:o.id,label:o[lang]}))}/>
        <DatePicker id="rem-date" yearly lang={lang} label={T.remDate} value={{cal:edit.cal,m:edit.m,d:edit.d}} onChange={v=>setEdit(e=>({...e,...v}))} disabled={!!(fixed||movable)} hint={fixed?T.fixedDate:movable?T.movableDate:undefined}
          labels={{jalali:T.jalali,gregorian:T.gregorian,month:T.month,day:T.dayL}}/>
        <Select id="rem-before" label={T.remBefore} value={String(edit.before)} onChange={e=>setEdit({...edit,before:+e.target.value})} options={[1,3,7].map(v=>({value:String(v),label:T.beforeOpt[v]}))}/>
        <div style={{display:'flex',flexDirection:'column',gap:6}}><span className="ag-field__label">{T.via}</span><Tabs variant="pill" items={[{id:'sms',label:T.sms},{id:'wa',label:T.waShort}]} value={edit.channel} onChange={v=>setEdit({...edit,channel:v})}/></div>
      </div>}
    </Dialog>
    {toast&&<div style={{position:'fixed',bottom:mobile?72:24,insetInline:mobile?'12px':'auto 24px',zIndex:120}}><Toast title={toast} onClose={()=>setToast('')} style={mobile?{width:'100%'}:undefined}/></div>}
  </div>;
}
Object.assign(window,{Reminders,normReminder});
